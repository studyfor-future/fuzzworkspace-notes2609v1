issue #197

There is a out bound read vulnerability in getNumberOfElements because getNumberOfElements doesn't check bufPos, elementLength arguments. So bufPos and elementLength can be very large number.

#include <stdio.h>
#include <stdint.h>
#include <unistd.h>
#include <libiec61850/mms_value.h>
uint8_t globalbuf[0x10000];
int main(){
    ssize_t readcount;
    readcount = read(0, globalbuf, sizeof(globalbuf));
    if( readcount <= 0 ){
        return 0;
    }
    MmsValue *res = MmsValue_decodeMmsData(globalbuf, 0, (int)readcount, 0);
    return 0;
}
